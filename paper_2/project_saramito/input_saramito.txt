
! input_saramito.txt

  meshfile = 'mesh.out'

  eta_s = 0.1    ! solvent viscosity

  model = 3      ! 2: UCM/Oldroyd-B, 3: Giesekus, 5: PTT linear 6: PTT exp

  Gmod = 0.5     ! G modulus

  tau_y = 2.0    ! yield stress for SRM1 and SRM2

  K = 1.0        ! power-law coefficient for SRM2
  n = 0.5        ! power-law exponent for SRM2

  force = 1.0    ! force on sphere

  deltat = 0.01       ! time step
  numtimesteps = 3000 ! number of time steps

  vtkevery = 50  ! vtk file every vtkevery steps. 0: means none

  real_storage_gup = 1.4     ! real_storage gup LU (HSL)
  integer_storage_gup = 1.4  ! integer_storage gup LU (HSL)
  real_storage_c  = 1.6      ! real_storage for the conformation LU (HSL)
  integer_storage_c  = 1.6   ! integer_storage for conformation LU (HSL)
